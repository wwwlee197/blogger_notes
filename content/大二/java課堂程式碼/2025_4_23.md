```java
package com.example.a2024_unit041_secondactivity;  
  
import android.app.Activity;  
import android.content.Intent;  
import android.os.Bundle;  
import android.view.View;  
import android.widget.Button;  
import android.widget.TextView;  
  
public class MainActivity extends Activity {  
  
    public static final int REQUEST_CODE_GAME = 1;  
    private TextView txtResult;  
  
    // 用來記錄勝負的局數  
    private int playerWins = 0;  
    private int computerWins = 0;  
    private int totalGames = 0;  
  
    @Override  
    protected void onCreate(Bundle savedInstanceState) {  
        super.onCreate(savedInstanceState);  
        setContentView(R.layout.activity_main); // 記得確認你的 layout 名稱  
  
        Button btnLaunchGame = findViewById(R.id.btnLaunchGame);  
        txtResult = findViewById(R.id.txtResult);  
  
        btnLaunchGame.setOnClickListener(new View.OnClickListener() {  
            @Override  
            public void onClick(View view) {  
                Intent it = new Intent(MainActivity.this, GameActivity.class);  
                startActivityForResult(it, REQUEST_CODE_GAME);  
            }  
        });  
    }  
  
    @Override  
    protected void onActivityResult(int requestCode, int resultCode, Intent data) {  
        super.onActivityResult(requestCode, resultCode, data);  
  
        if (requestCode == REQUEST_CODE_GAME && resultCode == RESULT_OK) {  
            String result = data.getStringExtra("gameResult");  
  
            // 根據結果更新勝負統計  
            if (result.contains("你贏了")) {  
                playerWins++;  
            } else if (result.contains("你輸了")) {  
                computerWins++;  
            }  
  
            totalGames++;  // 每次結束一局，總局數加1  
  
            // 顯示遊戲結果和統計信息  
            String stats = "總局數: " + totalGames + "\n" +  
                    "玩家勝利: " + playerWins + "\n" +  
                    "電腦勝利: " + computerWins + "\n\n" +  
                    "比賽結果：" + result;  
            txtResult.setText(stats);  
        }  
    }  
}
```

GameActivity.java
```java
package com.example.a2024_unit041_secondactivity;  
  
import android.content.Intent;  
import android.os.Bundle;  
import android.view.View;  
import android.widget.ImageButton;  
import android.widget.ImageView;  
import android.widget.TextView;  
  
import androidx.appcompat.app.AppCompatActivity;  
  
import java.util.Random;  
  
public class GameActivity extends AppCompatActivity {  
  
    private ImageView imgPcDice, imgMyDice;  
    private ImageButton imgStart;  
    private TextView txtResult;  
  
    private int[] playerDiceImages = {  
            R.drawable.b1, R.drawable.b2, R.drawable.b3,  
            R.drawable.b4, R.drawable.b5, R.drawable.b6  
    };  
  
    private int[] computerDiceImages = {  
            R.drawable.h1, R.drawable.h2, R.drawable.h3,  
            R.drawable.h4, R.drawable.h5, R.drawable.h6  
    };  
  
    @Override  
    protected void onCreate(Bundle savedInstanceState) {  
        super.onCreate(savedInstanceState);  
        setContentView(R.layout.activity_game);  
  
        imgPcDice = findViewById(R.id.img_pc_dice);  
        imgMyDice = findViewById(R.id.img_my_dice);  
        imgStart = findViewById(R.id.img_start);  
        txtResult = findViewById(R.id.textView2);  
  
        imgStart.setOnClickListener(new View.OnClickListener() {  
            @Override  
            public void onClick(View v) {  
                rollDices();  
            }  
        });  
    }  
  
    private void rollDices() {  
        txtResult.setText("電腦搖骰中...");  
        imgStart.setEnabled(false); // 防止連點  
  
        new Thread(new Runnable() {  
            @Override  
            public void run() {  
                Random random = new Random();  
                int finalPlayerRoll = random.nextInt(6);  
                int finalComputerRoll = 0;  
  
                int totalDuration = 2000; // 10秒  
                int interval = 100; // 每0.1秒更新一次  
                int totalSteps = totalDuration / interval;  
  
                for (int i = 0; i < totalSteps; i++) {  
                    final int tempRoll = random.nextInt(6);  
                    final int tempPlayer = random.nextInt(6);  
  
                    runOnUiThread(new Runnable() {  
                        @Override  
                        public void run() {  
                            imgPcDice.setImageResource(computerDiceImages[tempRoll]);  
                            imgMyDice.setImageResource(playerDiceImages[tempPlayer]);  
                        }  
                    });  
  
                    try {  
                        Thread.sleep(interval);  
                    } catch (InterruptedException e) {  
                        e.printStackTrace();  
                    }  
  
                    if (i == totalSteps - 1) {  
                        finalComputerRoll = tempRoll;  
                        finalPlayerRoll = tempPlayer;  
                    }  
                }  
  
                final int finalComp = finalComputerRoll;  
                final int finalPlayer = finalPlayerRoll;  
  
                runOnUiThread(new Runnable() {  
                    @Override  
                    public void run() {  
                        imgPcDice.setImageResource(computerDiceImages[finalComp]);  
                        imgMyDice.setImageResource(playerDiceImages[finalPlayer]);  
  
                        String resultText;  
                        if (finalPlayer > finalComp) {  
                            resultText = "你贏了！";  
                        } else if (finalPlayer < finalComp) {  
                            resultText = "你輸了！";  
                        } else {  
                            resultText = "平手！";  
                        }  
  
                        resultText += "\n你: " + (finalPlayer + 1) + "00點  |  電腦: " + (finalComp + 1) + "00點";  
  
                        // 傳回遊戲結果給 MainActivity                        Intent resultIntent = new Intent();  
                        resultIntent.putExtra("gameResult", resultText);  
                        setResult(RESULT_OK, resultIntent);  
                        finish();  
                    }  
                });  
            }  
        }).start();  
    }  
}
```
