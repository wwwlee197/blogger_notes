import rclpy
from rclpy.node import Node
import cv2
import numpy as np
import math
from cv2 import aruco
from std_msgs.msg import Int16MultiArray

# 匯入控制器
from controller import Controller

import vision_utils as vu

# 定義程式狀態
MODE_WAIT_FOR_TARGET = 1    # 等待目標模式
MODE_TRACKING = 2           # 追蹤模式

class SingleArucoMain_PRO(Node):
    def __init__(self, cam_index=4, npz_file='calibration_864x480.npz'):
        super().__init__('single_aruco_id2')

        # 攝影機初始化
        self.cam_index = cam_index
        self.cap = None
        self.cam_width = 845
        self.cam_height = 480

        # ROS 相關設定
        self.controller_00 = Controller(self, '/set_pwm')
        self.controller_01 = Controller(self, '/set_pwm_01')

        
        # 目標位置 (共享世界座標 xyz) 與對應像素位置
        self.target_global_pos = None
        self.target_pixel = None
        
        # ArUco 相關設定
        self.aruco_dict = aruco.getPredefinedDictionary(aruco.DICT_6X6_1000)
        self.aruco_params = aruco.DetectorParameters()
        self.marker_length = 0.09
        
        self.leader_id = 8           # 領航車 ArUco ID（leader twrb）
        self.follower_id = 12        # 跟隨車 ArUco ID（follower twrb）

        self.follow_distance = 0.30  # 自走車間距(公尺)
        self.hold_ms = 300           # leader 短暫丟失時的目標保持時間
        self.last_leader_pose = None
        self.last_leader_stamp = 0.0

        # 導入相機校正 npz 檔
        data = np.load(npz_file)
        self.camera_matrix = data['camMatrix']
        self.dist_coeffs = data['distCoeff']
        
        # 設定狀態
        self.mode = MODE_WAIT_FOR_TARGET

        print("Single-Cam Start")
        print("點擊畫面設定目標點")
        print("按 [ Enter ] 導航")
        print("按 [ ESC ] 離開")


        # 初始化攝影機，失敗則程式結束
        if not self.init_camera():
            return

        # 建立視窗顯示相機畫面
        self.win_name = f'Camera {self.cam_index} (Single Cam)'
        cv2.namedWindow(self.win_name, cv2.WINDOW_NORMAL)
        cv2.setMouseCallback(self.win_name, self.mouse_callback)

        # 定時器迴圈
        self.timer = self.create_timer(0.15, self.control_loop)
        
        # 確保程式不會因為初始化失敗而繼續執行
        self.is_ready = True
# ================================================= 節點初始化結束 =================================================
    
    def init_camera(self):
        print(f"初始化攝影機 {self.cam_index}")
        self.cap = cv2.VideoCapture(self.cam_index)
        
        # 檢查攝影機是否開啟成功
        if not self.cap.isOpened():
            self.get_logger().error(f"無法開啟攝影機 {self.cam_index}")
            self.clean_up()
            return False
        
        self.cap.set(cv2.CAP_PROP_FRAME_WIDTH, self.cam_width)
        self.cap.set(cv2.CAP_PROP_FRAME_HEIGHT, self.cam_height)
        
        self.cap.set(cv2.CAP_PROP_FPS, 60)
        self.get_logger().info("攝影機初始化成功")
        return True

    def pixel_to_world_coords(self, px, py):
        """將像素點轉換為簡化的 3D 世界座標（Z固定）。"""
        fx = self.camera_matrix[0, 0]
        fy = self.camera_matrix[1, 1]
        cx = self.camera_matrix[0, 2]
        cy = self.camera_matrix[1, 2]
        fixed_z = 2.1   # 相機到地面的距離z (公尺)
        target_x = (px - cx) * fixed_z / fx
        target_y = (py - cy) * fixed_z / fy
        return np.array([target_x, target_y, fixed_z])


    def mouse_callback(self, event, x, y, flags, param):
        if event == cv2.EVENT_LBUTTONDOWN and self.mode == MODE_WAIT_FOR_TARGET:
            # ---------------------------------------------------------
            # 1️. 使用者點目標 → 計算世界座標
            # ---------------------------------------------------------
            self.target_pixel = (x, y)
            self.target_global_pos = self.pixel_to_world_coords(x, y)

            print(f"  點擊像素座標: ({x}, {y})")
            print(f"  共享世界座標 (xyz): ({self.target_global_pos[0]:.2f}, {self.target_global_pos[1]:.2f}, {self.target_global_pos[2]:.2f})")

            # ---------------------------------------------------------
            # 2. 偵測目前畫面中的兩台車（ID8、ID12）的位置
            # ---------------------------------------------------------
            # 擷取畫面
            ret, frame = self.cap.read()
            if not ret:
                print("cap 畫面讀取失敗")
                return

            # 偵測 aruco
            corners, ids, _ = aruco.detectMarkers(frame, self.aruco_dict, parameters=self.aruco_params)

            if ids is None:
                print("未偵測到自走車")
                return

            ids_flat = ids.flatten()

            # 自走車 3D 座標位置
            pos8 = None
            pos12 = None

            # 取得 ID=8 的位置
            if 8 in ids_flat:
                idx8 = int(np.where(ids_flat == 8)[0][0])
                rvec8, tvec8, _ = aruco.estimatePoseSingleMarkers(
                    corners[idx8], self.marker_length, self.camera_matrix, self.dist_coeffs)
                pos8 = tvec8[0][0]   # 3D pos

            # 取得 ID=12 的位置
            if 12 in ids_flat:
                idx12 = int(np.where(ids_flat == 12)[0][0])
                rvec12, tvec12, _ = aruco.estimatePoseSingleMarkers(
                    corners[idx12], self.marker_length, self.camera_matrix, self.dist_coeffs)
                pos12 = tvec12[0][0]  # 3D pos

            # ---------------------------------------------------------
            # 3️. 比較兩台車與目標點位置 → 靠近目標的當 leader
            # ---------------------------------------------------------
            if pos8 is not None and pos12 is not None:

                d8 = np.linalg.norm((self.target_global_pos - pos8)[:2])
                d12 = np.linalg.norm((self.target_global_pos - pos12)[:2])

                if d8 < d12:
                    self.leader_id = 8
                    self.follower_id = 12
                    self.controller_00 = Controller(self, '/set_pwm')
                    self.controller_01 = Controller(self, '/set_pwm_01')
                    print("距離判定:ID 8 為 Leader,ID 12 為 Follower")
                else:
                    self.leader_id = 12
                    self.follower_id = 8
                    self.controller_00 = Controller(self, '/set_pwm_01')
                    self.controller_01 = Controller(self, '/set_pwm') 
                    print("距離判定:ID 12 為 Leader,ID 8 為 Follower")

            else:
                print("判定失敗，重新點擊畫面")


            print("----> 按下「Enter」鍵開始追蹤")


    # 計算「距離 d、位於兩車之間」的目標
    def compute_mid_target(self, leader_pos, follower_pos, d): 
        # 向量V: 從 Follower 指向 Leader 的方向
        v = leader_pos - follower_pos
        # 只用水平距離（x,y），忽略 z 抖動，計算兩車距離 L
        L = np.linalg.norm(v[:2])
        if L < 1e-6:
            return None, L
        vhat = v / np.linalg.norm(v)    #單位向量 vhat = v 的方向，但長度 = 1
        target = leader_pos - d * vhat  # 距離 leader 為 d 的點（在兩車之間）
        return target, L

# ================================================= twrb 控制迴圈 =================================================
    def control_loop(self):
        # 檢查程式是否因初始化失敗而中斷
        if not hasattr(self, 'is_ready') or not self.is_ready:
            return

        ret, frame = self.cap.read()
        if not ret:
            self.get_logger().error("無法讀取攝影機畫面，程式將結束。")
            self.clean_up()
            rclpy.shutdown()
            return
            
        disp = frame.copy()

        # -----------------------------------------------------------
        # 1. 偵測 ArUco 標籤
        # -----------------------------------------------------------
        corners, ids, _ = aruco.detectMarkers(frame, self.aruco_dict, parameters=self.aruco_params)
        
        # 繪製偵測結果
        if ids is not None:
            aruco.drawDetectedMarkers(disp, corners, ids)

        # -----------------------------------------------------------
        # 2. 處理不同模式下的邏輯
        # -----------------------------------------------------------
        if self.mode == MODE_WAIT_FOR_TARGET:
            # 若已有目標，畫出目標點
            if self.target_pixel is not None:
                cv2.circle(disp, self.target_pixel, 4, (0, 0, 255), -1)  # 紅點
        
        elif self.mode == MODE_TRACKING:
            leader_pos, leader_orient = None, None
            follower_pos, follower_orient = None, None

            if ids is not None:
                ids_flat = ids.flatten()

                # 取 leader 位姿
                if ids is not None and self.leader_id in ids:
                    idx = int(np.where(ids.flatten() == self.leader_id)[0][0])
                    rvec, tvec, _ = aruco.estimatePoseSingleMarkers(
                        corners[idx], self.marker_length, self.camera_matrix, self.dist_coeffs)
                    
                    leader_pos = tvec[0][0]  # 相機座標系下的 3D 位置
                    rmat, _ = cv2.Rodrigues(rvec)
                    y_axis = rmat @ np.array([0, 1, 0])
                    leader_orient = math.atan2(y_axis[0], y_axis[1])
                    cv2.drawFrameAxes(disp, self.camera_matrix, self.dist_coeffs, rvec, tvec.reshape(3, 1), self.marker_length * 0.5)
                    
                    # 記錄可短暫保形
                    self.last_leader_pose = (leader_pos.copy(), leader_orient)
                    self.last_leader_stamp = cv2.getTickCount() / cv2.getTickFrequency()


                # 取 follower 位姿
                if ids is not None and self.follower_id in ids:
                    idx = int(np.where(ids.flatten() == self.follower_id)[0][0])
                    rvec, tvec, _ = aruco.estimatePoseSingleMarkers(
                        corners[idx], self.marker_length, self.camera_matrix, self.dist_coeffs)
                    
                    follower_pos = tvec[0][0]  # 相機座標系下的 3D 位置
                    rmat, _ = cv2.Rodrigues(rvec)
                    y_axis = rmat @ np.array([0, 1, 0])
                    follower_orient = math.atan2(y_axis[0], y_axis[1])
                    cv2.drawFrameAxes(disp, self.camera_matrix, self.dist_coeffs, rvec, tvec.reshape(3, 1), self.marker_length * 0.5)

            # ----------------- Leader：滑鼠點目標 → PWM -----------------
            if leader_pos is not None and leader_orient is not None and self.target_global_pos is not None:
                pwmL = self.controller_00.compute_pwm(leader_pos, leader_orient, self.target_global_pos) # 導航
                self.controller_00.pwm_pub.publish(Int16MultiArray(data=pwmL))
            else:
                self.controller_00.pwm_pub.publish(Int16MultiArray(data=[0, 0])) # 若無法追蹤則停止
            
            # ----------------- Follower：兩車之間距離 d → PWM -----------------
            follower_target, L = None, None
            now = cv2.getTickCount() / cv2.getTickFrequency()

            # leader 可見：即時計算
            if leader_pos is not None and follower_pos is not None:
                follower_target, L = self.compute_mid_target(leader_pos, follower_pos, self.follow_distance)
                # 讓靠近時角度對齊（追尾更安全）
             
                self.controller_01.target_orientation = leader_orient

            # leader 短暫丟失：用最後一次的 leader 位姿保形 hold_ms
            elif (self.last_leader_pose is not None and follower_pos is not None and
                (now - self.last_leader_stamp) * 1000.0 < self.hold_ms):
                last_pos, last_theta = self.last_leader_pose
                follower_target, L = self.compute_mid_target(last_pos, follower_pos, self.follow_distance)
                self.controller_01.target_orientation = last_theta

            # 發 follower 的 PWM（含安全距離保護）
            if follower_pos is not None and follower_target is not None and follower_orient is not None:
                # 若距離已小於 d*0.9，就停車（避免追尾）；保角度由 controller 內部完成
                if L is not None and L <= self.follow_distance * 0.9:
                    pwmF = [0, 0]
                else:
                    pwmF = self.controller_01.compute_pwm(follower_pos, follower_orient, follower_target)
                self.controller_01.pwm_pub.publish(Int16MultiArray(data=pwmF))
                #（可選）畫出 follower 目標點
                img_pt, _ = cv2.projectPoints(follower_target.reshape(1,1,3),
                                            np.zeros((3,1)), np.zeros((3,1)),
                                            self.camera_matrix, self.dist_coeffs)
                cv2.circle(disp, tuple(img_pt[0,0].astype(int)), 6, (0,255,0), 2)
            else:
                self.controller_01.pwm_pub.publish(Int16MultiArray(data=[0, 0]))

            # 繪製滑鼠目標點（leader 用）
            if self.target_pixel is not None:
                cv2.circle(disp, self.target_pixel, 8, (0, 0, 255), -1)  # 紅點 
        
        # -----------------------------------------------------------
        # 3. 顯示畫面與鍵盤處理
        # -----------------------------------------------------------
        cv2.imshow(self.win_name, disp)
        
        key = cv2.waitKey(1) & 0xFF
        if key == 27:  # ESC 鍵
            print("按下 ESC 結束程式")
            self.clean_up()
            rclpy.shutdown()
        elif key == 13: # Enter 鍵
            if self.mode == MODE_WAIT_FOR_TARGET and self.target_global_pos is not None:
                self.mode = MODE_TRACKING
                print("\n目的地設定成功,請繫好安全帶")
            elif self.mode == MODE_TRACKING:
                print("移動中...")
            else:
                print("---- 請先設定目標點 ----")

    # 結束程式 停止自走車  
    def clean_up(self):
        self.controller_00.pwm_pub.publish(Int16MultiArray(data=[0, 0])) # 停止自走車
        self.controller_01.pwm_pub.publish(Int16MultiArray(data=[0, 0])) 
        if self.cap:
            self.cap.release()
        cv2.destroyAllWindows()
        print("車輛已達目的地,請解開安全帶,下車時注意後方來車,祝您旅途愉快～")

from std_msgs.msg import Int16MultiArray
import numpy as np
import math

class Controller:
    def __init__(self, node, topic):
        self.node = node
        # 建立 publish_node 
        self.pwm_pub = node.create_publisher(Int16MultiArray, topic, 5)

        # PID 演算法 Kp-比例增益    Ki-積分增益    Kd-微分增益
        # 控制機器人進退的「距離」誤差
        self.Kp_linear = 20.0
        self.Ki_linear = 0.1
        self.Kd_linear = 20.0

        # 控制機器人旋轉的「角度」誤差
        self.Kp_angular = 10.0
        self.Ki_angular = 0.01
        self.Kd_angular = 30.0

        # 計算誤差
        self.prev_err_dis = 0.0     # 上一次的距離誤差
        self.prev_err_theta = 0.0   # 上一次的角度誤差
        self.integral_dis = 0.0     # 距離誤差的積分值
        self.integral_theta = 0.0   # 角度誤差的積分值

        self.max_pwm = 50       # 馬達最 大 值 
        self.deadzone = 30      # 馬達最 小 值

        self.marker_length = 0.09
        
        self.robot_id = 2
        self.target_id = 1
        self.aligning = False
        self.target_orientation = None
        self.stop_sent = False  # ✅ 加入對齊停止旗標
    
    # 輸出控制左右輪馬達pwm
    def compute_pwm(self, robot_pos, robot_orient, target_pos):
        # robot_pos     機器人當前位置 
        # robot_orient  機器人當前朝向  
        # target_pos    目標點位置
        if robot_pos is None or robot_orient is None or target_pos is None:
            return [0, 0]

        dx = target_pos[0] - robot_pos[0]
        dy = target_pos[1] - robot_pos[1]
        err_dis = math.sqrt(dx ** 2 + dy ** 2)

        # 接近目標時候 專注於角度調整
        if err_dis < 0.1:
            if self.target_orientation is not None:
                err_theta = (self.target_orientation - robot_orient + math.pi) % (2 * math.pi) - math.pi
                self.integral_theta += err_theta
                derivative_theta = err_theta - self.prev_err_theta
                self.prev_err_theta = err_theta

                angular_output = (self.Kp_angular * err_theta +
                                  self.Ki_angular * self.integral_theta +
                                  self.Kd_angular * derivative_theta)
                linear_output = 0.0

                if abs(err_theta) < 0.75:
                    if not self.stop_sent:
                        print("✅ ID2 對齊完成，傳送一次停止PWM")
                        self.stop_sent = True
                        return [0, 0]
                    else:
                        return [0, 0]
            else:
                return [0, 0]
        # 距離較遠時 先轉想 後前進
        else:
            self.stop_sent = False  # ✅ 若有移動，重新允許送停止PWM

            target_angle = math.atan2(dx, dy)
            err_theta = (target_angle - robot_orient + math.pi) % (2 * math.pi) - math.pi

            self.integral_dis += err_dis
            self.integral_theta += err_theta

            derivative_dis = err_dis - self.prev_err_dis
            derivative_theta = err_theta - self.prev_err_theta
            self.prev_err_dis = err_dis
            self.prev_err_theta = err_theta

            linear_output = self.Kp_linear * err_dis + self.Ki_linear * self.integral_dis + self.Kd_linear * derivative_dis
            angular_output = self.Kp_angular * err_theta + self.Ki_angular * self.integral_theta + self.Kd_angular * derivative_theta

            if abs(err_theta) > 0.3:
                linear_output = 0.0

        left_pwm = linear_output - angular_output
        right_pwm = linear_output + angular_output

        left_pwm = int(max(min(left_pwm, self.max_pwm), -self.max_pwm))
        right_pwm = int(max(min(right_pwm, self.max_pwm), -self.max_pwm))

        if 0 < abs(left_pwm) < self.deadzone:
            left_pwm = int(math.copysign(self.deadzone, left_pwm))
        if 0 < abs(right_pwm) < self.deadzone:
            right_pwm = int(math.copysign(self.deadzone, right_pwm))

        return [left_pwm, right_pwm]
